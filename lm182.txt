function geoq_lm_code() {
	$chars = 'abcdefghjkmnpqrstuvwxyz23456789';
	$max   = strlen( $chars ) - 1;
	do {
		$code = '';
		for ( $i = 0; $i < 8; $i++ ) {
			$code .= $chars[ random_int( 0, $max ) ];
		}
	} while ( geoq_lm_find( $code ) );
	return $code;
}

function geoq_lm_find( $code ) {
	$q = get_posts( array(
		'post_type'      => 'geoq_livemug',
		'post_status'    => 'any',
		'meta_key'       => '_lm_code',
		'meta_value'     => $code,
		'posts_per_page' => 1,
		'fields'         => 'ids',
	) );
	return $q ? (int) $q[0] : 0;
}

function geoq_lm_url( $att_id ) {
	return $att_id ? wp_get_attachment_url( (int) $att_id ) : '';
}

// Allow the AR target file type only while our endpoint is saving files.
$GLOBALS['geoq_lm_saving'] = false;
add_filter( 'upload_mimes', function ( $m ) {
	if ( ! empty( $GLOBALS['geoq_lm_saving'] ) ) {
		$m['mind'] = 'application/octet-stream';
	}
	return $m;
} );
add_filter( 'wp_check_filetype_and_ext', function ( $data, $file, $filename ) {
	if ( ! empty( $GLOBALS['geoq_lm_saving'] ) && preg_match( '/\.mind$/i', $filename ) ) {
		$data['ext']  = 'mind';
		$data['type'] = 'application/octet-stream';
		$data['proper_filename'] = $filename;
	}
	return $data;
}, 10, 3 );

function geoq_lm_save_file( $key, $post_id, $types, $max_mb ) {
	if ( empty( $_FILES[ $key ] ) || ! empty( $_FILES[ $key ]['error'] ) ) {
		return 0;
	}
	$f = $_FILES[ $key ];
	if ( $f['size'] > $max_mb * 1048576 ) {
		return new WP_Error( 'too_big', $key . ' is too large', array( 'status' => 413 ) );
	}
	$ok = false;
	foreach ( $types as $t ) {
		if ( strpos( (string) $f['type'], $t ) === 0 ) {
			$ok = true;
		}
	}
	if ( ! $ok ) {
		return new WP_Error( 'bad_type', $key . ' has a wrong file type', array( 'status' => 415 ) );
	}
	require_once ABSPATH . 'wp-admin/includes/file.php';
	require_once ABSPATH . 'wp-admin/includes/media.php';
	require_once ABSPATH . 'wp-admin/includes/image.php';
	$GLOBALS['geoq_lm_saving'] = true;
	$id = media_handle_upload( $key, $post_id, array(), array( 'test_form' => false ) );
	$GLOBALS['geoq_lm_saving'] = false;
	return $id;
}

function geoq_lm_rate_ok() {
	$ip  = isset( $_SERVER['REMOTE_ADDR'] ) ? sanitize_text_field( wp_unslash( $_SERVER['REMOTE_ADDR'] ) ) : 'x';
	$key = 'geoq_lm_rl_' . md5( $ip );
	$n   = (int) get_transient( $key );
	if ( $n >= 12 ) {
		return false;
	}
	set_transient( $key, $n + 1, HOUR_IN_SECONDS );
	return true;
}

function geoq_lm_create( WP_REST_Request $r ) {
	if ( ! geoq_lm_rate_ok() ) {
		return new WP_Error( 'slow_down', 'Too many uploads. Please try again later.', array( 'status' => 429 ) );
	}
	if ( $r->get_param( 'agree' ) !== '1' ) {
		return new WP_Error( 'rules', 'Please confirm the content rules.', array( 'status' => 400 ) );
	}
	$email = sanitize_email( (string) $r->get_param( 'email' ) );
	if ( ! is_email( $email ) ) {
		return new WP_Error( 'email', 'Please enter a valid email.', array( 'status' => 400 ) );
	}
	if ( empty( $_FILES['photo'] ) || empty( $_FILES['video'] ) || empty( $_FILES['mind'] ) ) {
		return new WP_Error( 'files', 'Photo and video are required.', array( 'status' => 400 ) );
	}
	$code = geoq_lm_code();
	$pid  = wp_insert_post( array(
		'post_type'   => 'geoq_livemug',
		'post_status' => 'publish',
		'post_title'  => 'Live Mug ' . $code . ' — ' . $email,
	), true );
	if ( is_wp_error( $pid ) ) {
		return $pid;
	}
	$files = array(
		'photo' => array( array( 'image/' ), 15 ),
		'video' => array( array( 'video/' ), 95 ),
		'mind'  => array( array( 'application/octet-stream', '' ), 6 ),
		'logo'  => array( array( 'image/' ), 6 ),
	);
	$ids = array();
	foreach ( $files as $k => $rule ) {
		$id = geoq_lm_save_file( $k, $pid, $rule[0], $rule[1] );
		if ( is_wp_error( $id ) ) {
			wp_delete_post( $pid, true );
			return $id;
		}
		$ids[ $k ] = (int) $id;
	}
	if ( ! $ids['photo'] || ! $ids['video'] || ! $ids['mind'] ) {
		wp_delete_post( $pid, true );
		return new WP_Error( 'upload', 'Upload failed. Please try again.', array( 'status' => 500 ) );
	}
	$gift = json_decode( (string) $r->get_param( 'gift' ), true );
	$meta = array(
		'_lm_code'    => $code,
		'_lm_token'   => wp_generate_password( 24, false ),
		'_lm_email'   => $email,
		'_lm_qty'     => ( $r->get_param( 'qty' ) === '2' ) ? 2 : 1,
		'_lm_logo'    => $ids['logo'] ? 1 : 0,
		'_lm_site'    => sanitize_text_field( mb_substr( (string) $r->get_param( 'site' ), 0, 32 ) ),
		'_lm_edit'    => ( $r->get_param( 'edit' ) === '1' ) ? 1 : 0,
		'_lm_lang'    => sanitize_key( (string) $r->get_param( 'lang' ) ),
		'_lm_gift'    => is_array( $gift ) ? array_map( 'sanitize_text_field', $gift ) : array(),
		'_lm_status'  => 'new',
		'_lm_photo'   => $ids['photo'],
		'_lm_video'   => $ids['video'],
		'_lm_mind'    => $ids['mind'],
		'_lm_logo_id' => $ids['logo'],
	);
	foreach ( $meta as $k => $v ) {
		update_post_meta( $pid, $k, $v );
	}
	$live = home_url( '/live/' . $code . '/' );
	$admin = admin_url( 'post.php?post=' . $pid . '&action=edit' );
	wp_mail(
		get_option( 'admin_email' ),
		'New Live Mug design ' . $code,
		"A customer finished a Live Mug design (checkout may still be in progress).\n\nCode: " . $code . "\nEmail: " . $email . "\nMugs: " . $meta['_lm_qty'] . "\nAR page: " . $live . "\nReview and print file: " . $admin . "\n\nPlease watch the video before ordering the print."
	);
	return array( 'code' => $code, 'live' => $live );
}

function geoq_lm_replace_video( WP_REST_Request $r ) {
	$pid = geoq_lm_find( (string) $r['code'] );
	if ( ! $pid || ! hash_equals( (string) get_post_meta( $pid, '_lm_token', true ), (string) $r->get_param( 't' ) ) ) {
		return new WP_Error( 'denied', 'This link is not valid.', array( 'status' => 403 ) );
	}
	if ( ! get_post_meta( $pid, '_lm_edit', true ) ) {
		return new WP_Error( 'locked', 'Full editing is not included with this mug.', array( 'status' => 403 ) );
	}
	if ( ! geoq_lm_rate_ok() ) {
		return new WP_Error( 'slow_down', 'Too many uploads. Please try again later.', array( 'status' => 429 ) );
	}
	$id = geoq_lm_save_file( 'video', $pid, array( 'video/' ), 95 );
	if ( is_wp_error( $id ) || ! $id ) {
		return is_wp_error( $id ) ? $id : new WP_Error( 'upload', 'Upload failed.', array( 'status' => 500 ) );
	}
	$old = (int) get_post_meta( $pid, '_lm_video', true );
	update_post_meta( $pid, '_lm_video', (int) $id );
	update_post_meta( $pid, '_lm_video_changed', current_time( 'mysql' ) );
	if ( $old ) {
		wp_delete_attachment( $old, true );
	}
	wp_mail( get_option( 'admin_email' ), 'Live Mug video changed ' . (string) $r['code'], 'The buyer uploaded a new video. Please watch it: ' . admin_url( 'post.php?post=' . $pid . '&action=edit' ) );
	return array( 'ok' => true );
}

// Public AR page for one mug: /live/CODE/ and the edit page /live/CODE/edit/?t=TOKEN.
add_action( 'template_redirect', function () {
	$code = get_query_var( 'geoq_live' );
	if ( ! $code ) {
		return;
	}
	if ( ! defined( 'DONOTCACHEPAGE' ) ) {
		define( 'DONOTCACHEPAGE', true );
	}
	nocache_headers();
	$pid = geoq_lm_find( $code );
	if ( ! $pid || get_post_meta( $pid, '_lm_status', true ) === 'blocked' ) {
		status_header( 404 );
		header( 'Content-Type: text/html; charset=utf-8' );
		echo '<!DOCTYPE html><meta name="viewport" content="width=device-width,initial-scale=1"><title>GeoQartuli</title><body style="font:16px system-ui;background:#0B1530;color:#F5F0E9;display:grid;place-items:center;min-height:100vh;margin:0;text-align:center"><div><p>This mug link is not available.</p><p><a style="color:#E0C58F" href="https://geoqartuli.com">geoqartuli.com</a></p></div>';
		exit;
	}
	if ( get_query_var( 'geoq_live_edit' ) ) {
		geoq_lm_render_edit( $pid, $code );
		exit;
	}
	$html = get_post_field( 'post_content', GEOQ_LM_AR_PAGE, 'raw' );
	$base = 'https://giorgisheri.github.io/geoqartuli-ar/';
	$photo = wp_get_attachment_image_url( (int) get_post_meta( $pid, '_lm_photo', true ), 'medium' );
	$swap = array(
		$base . 'logo.mind' => geoq_lm_url( get_post_meta( $pid, '_lm_mind', true ) ),
		$base . 'promo.mp4' => geoq_lm_url( get_post_meta( $pid, '_lm_video', true ) ),
		$base . 'full.mp4'  => geoq_lm_url( get_post_meta( $pid, '_lm_video', true ) ),
		$base . 'logo.png'  => $photo ? $photo : geoq_lm_url( get_post_meta( $pid, '_lm_photo', true ) ),
	);
	header( 'Content-Type: text/html; charset=utf-8' );
	echo str_replace( array_keys( $swap ), array_values( $swap ), $html );
	exit;
}, 1 );

add_action( 'save_post_geoq_livemug', function ( $pid ) {
	if ( empty( $_POST['geoq_lm_nonce'] ) || ! wp_verify_nonce( sanitize_text_field( wp_unslash( $_POST['geoq_lm_nonce'] ) ), 'geoq_lm_save' ) || ! current_user_can( 'edit_post', $pid ) ) {
		return;
	}
	$new = isset( $_POST['geoq_lm_status'] ) ? sanitize_key( wp_unslash( $_POST['geoq_lm_status'] ) ) : '';
	if ( ! in_array( $new, array( 'new', 'approved', 'ordered', 'blocked' ), true ) ) {
		return;
	}
	$old = get_post_meta( $pid, '_lm_status', true );
	update_post_meta( $pid, '_lm_status', $new );
	if ( $new === 'ordered' && $old !== 'ordered' ) {
		$code = get_post_meta( $pid, '_lm_code', true );
		$msg  = "Your GeoQartuli Live Mug is being printed and will ship in 3–5 days.\n\nWatch your video on the mug: " . home_url( '/live/' . $code . '/' ) . "\n";
		if ( get_post_meta( $pid, '_lm_edit', true ) ) {
			$msg .= "\nChange your video any time (keep this link private): " . home_url( '/live/' . $code . '/edit/?t=' . get_post_meta( $pid, '_lm_token', true ) ) . "\n";
		}
		$msg .= "\nThank you!\nGeoQartuli · www.geoqartuli.com";
		wp_mail( get_post_meta( $pid, '_lm_email', true ), 'Your Live Mug is on its way', $msg );
	}
